## 五、生产环境部署

### 5.1 多模型路由

```python
# 多模型路由示例
from typing import Optional
import litellm

class MultiModelRouter:
    def __init__(self):
        self.models = {
            "fast": "gemini/gemini-2.5-flash",
            "balanced": "openai/gpt-5-turbo",
            "thinking": "anthropic/claude-3.7-sonnet",
        }
    
    async def route(self, task: str, requirement: str) -> str:
        """根据任务类型路由到最佳模型"""
        if "code" in requirement:
            return self.models["balanced"]
        elif "reasoning" in requirement:
            return self.models["thinking"]
        else:
            return self.models["fast"]
    
    async def generate(self, task: str, requirement: str) -> str:
        model = await self.route(task, requirement)
        response = await litellm.acompletion(
            model=model,
            messages=[{"role": "user", "content": task}]
        )
        return response.choices[0].message.content
```
